import hashlib
import json
import shutil
from pathlib import Path

              else 'context_scope' if 'outside' in case['id'] else 'pinyin_sandhi')
    rows.append(dict(id=case['id'],text=case['text'],blocked=level=='HIGH',
        expected_word=word,expected_level=level,
        expected_hits=[dict(word=word,level=level)] if word else [],category=category,
        kind='synthetic_development',label_source='expected mappings fixed before context revision; all84 original probes retained'))
fixture=ROOT/'docs/qa/data/us24-development.v3_11.jsonl'
with fixture.open('x',encoding='utf-8',newline='\n') as handle:
