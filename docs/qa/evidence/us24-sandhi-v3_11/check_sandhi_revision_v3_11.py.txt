def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()

def measurements(rows):
    result={}
    for mode in ['any_level','high_blocking']:
        matrix=dict(tp=0,fp=0,fn=0,tn=0)
        for row in rows:
            gold=bool(row['expected_hits']) if mode=='any_level' else row['blocked']
            if row['error']:
                predicted=not gold
            else:
                predicted=bool(row['response']['hits']) if mode=='any_level' else not row['response']['passed']
            matrix['tp' if predicted and gold else 'fp' if predicted else 'fn' if gold else 'tn']+=1
        tp,fp,fn,tn=(matrix[k] for k in ['tp','fp','fn','tn'])
        result[mode]=dict(confusion_matrix=matrix,recall=tp/(tp+fn) if tp+fn else None,
            precision=tp/(tp+fp) if tp+fp else None,fpr=fp/(fp+tn) if fp+tn else None)
    return result
