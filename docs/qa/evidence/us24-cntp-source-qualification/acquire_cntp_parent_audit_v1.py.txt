import hashlib
import json
import re
import urllib.request
from concurrent.futures import ThreadPoolExecutor
from datetime import UTC,datetime
from pathlib import Path

LAB=Path.cwd()/'qa-private/us24-recall-20261124'
REV='b1e6922487785aec59841bfb909bf40326174373'
ITEMS=[('CNTP_dataset/toxic_base.json',490933,'861a08bf50a4aebb528fb45c7b68a935786e998e'),
       ('CNTP_dataset/non_toxic_base.json',364607,'8543b162b5a571e315b5a499ad1ff0cd3722f2d0'),
       ('CNTP_dataset/perturbed_data/homo_toxi_transform.json',969880,'5542f148f55ce8434a95b7340ad42d4652999eeb'),
       ('LICENSE',1070,'4b10005c9b8c4f235fcd6c6fcfcf91d327f74ff7')]
OUT=LAB/'cntp-parent-audit-v1';OUT.mkdir(exist_ok=False)
policy=dict(created_at_utc=datetime.now(UTC).isoformat(),repo='thomasyyyoung/ToxiBenchCN',revision=REV,
    paper='https://arxiv.org/html/2505.24341v1',
    parent_provenance='paper4.1: baseToxi_CN,2533toxic2696non-toxic; no COLD derivative stated',
    scope='parent mapping qualification only; not training or fresh independent evaluation',
    admission='accept for later training only after explicit parent maps into existing training partition; retain ALL existing evaluation rows',
    redistribution='raw corpora remain private; preserve parent ToxiCN CC BY-NC-ND4.0 and repository license fingerprints',
    COLD_dev_fetched=False,accuracy='NOT_RUN',production_acceptance=False)
with (OUT/'acquisition-policy.json').open('x',encoding='utf-8') as f:json.dump(policy,f,ensure_ascii=False,indent=2)


def get(item):
    remote,size,expected=item
    url='https://raw.githubusercontent.com/thomasyyyoung/ToxiBenchCN/'+REV+'/'+remote
    request=urllib.request.Request(url,headers={'User-Agent':'OUC-US24-parent-provenance-audit'})
    with urllib.request.urlopen(request,timeout=45) as response:data=response.read(size+1)
    assert len(data)==size,(remote,len(data))
    blob=hashlib.sha1(b'blob '+str(len(data)).encode()+b'\0'+data).hexdigest()
    assert blob==expected,(remote,blob)
    with (OUT/Path(remote).name).open('xb') as f:f.write(data)
    entry=dict(file=Path(remote).name,upstream_path=remote,bytes=len(data),git_blob_sha1=blob,
               sha256=hashlib.sha256(data).hexdigest())
    if remote.endswith('.json'):
        value=json.loads(data)
        entry.update(root_type=type(value).__name__,rows=len(value))
        row=value[0] if isinstance(value,list) else next(iter(value.values()))
        entry['first_row_type']=type(row).__name__
        if isinstance(row,dict):entry['field_names']=[k for k in row if re.fullmatch(r'[A-Za-z_][A-Za-z_0-9]{0,40}',k)]
    else:
        assert data.startswith(b'MIT License')
        entry['declared_license']='MIT'
    return entry


with ThreadPoolExecutor(max_workers=4) as pool:results=list(pool.map(get,ITEMS))
with (OUT/'source-manifest.json').open('x',encoding='utf-8') as f:json.dump(dict(policy=policy,files=results),f,ensure_ascii=False,indent=2)
print(json.dumps(results,ensure_ascii=False),flush=True)
