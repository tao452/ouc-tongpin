def write(name,data,origin=None):
    text=data.decode('utf-8-sig').replace('\r\n','\n')
    for source,label in [(ROOT,'<workspace>'),(ROOT.parent,'<workspace-parent>'),(Path.home(),'<userhome>')]:
        for prefix in sorted({str(source),str(source).replace('\\','/'),str(source).replace('/','\\')},key=len,reverse=True):
            text=text.replace(prefix,label)
    data=('\n'.join(line.rstrip(' \t') for line in text.split('\n')).rstrip('\n')+'\n').encode()
    assert len(data)<5*1024*1024
    with (public/name).open('xb') as handle:handle.write(data)
    value=dict(file=name,public_lf_sha256=hashlib.sha256(data).hexdigest())
    if origin:value.update(private_sha256=hashlib.sha256(origin.read_bytes()).hexdigest(),normalization='local paths, LF and terminal whitespace only; numeric outcomes unchanged')
    files.append(value)
