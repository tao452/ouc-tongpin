import hashlib
import json
from pathlib import Path

from pypinyin import Style,pinyin

ROOT=Path.cwd();LAB=ROOT/'qa-private/us24-recall-20261124'
policy=json.loads((LAB/'alias-phrases-v3_12-policy-r2.json').read_text(encoding='utf-8'))
entries=json.loads((ROOT/'services/algo/app/data/moderation-v3_12-candidate.json').read_text(encoding='utf-8'))
lookup={r['word']:r for r in entries}
rows=[]

def add(identifier,text,word,category):
    level=lookup[word]['level'] if word else ''
    rows.append(dict(id=identifier,text=text,expected_word=word,expected_level=level,
        expected_hits=[dict(word=word,level=level)] if word else [],blocked=level=='HIGH',
        category=category,kind='synthetic_development'))

for number,item in enumerate(policy['mappings'],1):
    word=item['word'];stem=f'AP312-{number:03d}'
    phonemes=' '.join(v[0] for v in pinyin(item['alias'],style=Style.NORMAL,heteronym=False))
    readings=[phonemes,item['primary_numeric'],*item['explicit_extra_tones']]
    for index,reading in enumerate(readings):
        base=stem+f'-R{index}'
        positive={'continuous':reading.replace(' ',''),'spaced':reading,
                  'punctuated':reading.replace(' ','.'),'uppercase':reading.upper(),
                  'fullwidth':''.join(chr(ord(c)+0xFEE0) if '!'<=c<='~' else c for c in reading)}
        for name,text in positive.items():add(base+'-'+name,text,word,'alias_phonetic_mapping')
        for name,text in {
            'identifier-prefix':'prefix_'+reading.replace(' ',''),
            'identifier-suffix':reading.replace(' ','')+'_suffix',
            'english-prefix':'prefix'+reading.replace(' ',''),
            'english-suffix':reading.replace(' ','')+'suffix',
            'incomplete':' '.join(reading.split()[:-1]),
            'too-wide-gap':reading.replace(' ','-'*9),
        }.items():add(base+'-'+name,text,'','phonetic_boundary')
        for name,text in {
            'report':'举报：“'+reading+'”请处理',
            'threat-report':'有人威胁我：“'+reading+'”让我很害怕',
            'language':'语言科普：“'+reading+'”是攻击表达',
            'discourage':'请不要说“'+reading+'”伤害同学',
        }.items():add(base+'-'+name,text,'','precise_context')
        add(base+'-outside-attack','举报：“'+reading+'”请处理。杀了你','杀了你','context_scope')
        add(base+'-dedup',reading+'。'+reading,word,'deduplication')
        if any(c.isdigit() for c in reading):
            first=next(c for c in reading if c.isdigit())
            wrong=reading.replace(first,'0',1)
            add(base+'-invalid-tone',wrong,'','phonetic_boundary')
        if word in ['泼你硫酸','捅你一刀']:
            add(base+'-denial','我不会'+reading,'','denial_scope')
            add(base+'-denial-reversal','我不会'+reading+'才怪',word,'denial_reversal')
    add(stem+'-hanzi',item['alias'],word,'alias_hanzi_regression')
for index,text in enumerate([
    'async await import Unicode normalize callback stack pointer thread pool',
    'wo3 xiang3 qu4 tu2 shu1 guan3 fu4 xi2',
    '明天去图书馆复习，约好一起做题。',
    '怎么打断断点调试？给函数传一个参数。',
    '全宿舍都在复习，全班都参加活动。',
    '化学课介绍硫酸的性质，实验要遵守操作要求。',
    '最近被同学排挤，希望有人陪我聊聊。',
    '我不想活了，希望能得到帮助。',
],1):add(f'AP312-N{index:03d}',text,'','benign_campus_help')
assert len({r['id'] for r in rows})==len(rows)
path=ROOT/'docs/qa/data/us24-development.v3_12.jsonl'
with path.open('x',encoding='utf-8',newline='\n') as handle:
    for row in rows:handle.write(json.dumps(row,ensure_ascii=False)+'\n')
print(json.dumps(dict(rows=len(rows),positive=sum(bool(r['expected_hits']) for r in rows),negative=sum(not r['expected_hits'] for r in rows),
    sha256=hashlib.sha256(path.read_bytes()).hexdigest())),flush=True)
