import hashlib
import json
from datetime import UTC, datetime
from pathlib import Path

ROOT=Path.cwd();cases=[]
variants={77:('你个脑瘫',['你个脑摊','你这脑摊','你个挠摊']),78:('你个智障',['你个智瘴','你个知障'])}


def add(identifier,text,word='',level='',kind='normal_context'):
    cases.append(dict(id=identifier,text=text,expected_word=word,expected_level=level,
        blocked=level=='HIGH',kind=kind,category='normal_context' if not word else kind))


for candidate,(word,aliases) in variants.items():
    for n,alias in enumerate(aliases,1):
        prefix=f'MEDCTX-{candidate:03d}-A{n}'
        add(prefix+'-direct',alias,word,'MEDIUM','mapped_hanzi_variant')
        for q,(left,right) in enumerate([('“','”'),('「','」'),('"','"'),("'","'"),('‘','’')],1):
            report=f'举报：{left}{alias}{right}请处理'
            add(prefix+f'-report-{q}',report)
            add(prefix+f'-continued-{q}',report+'。'+alias,word,'MEDIUM','continued_attack')
for a,subject in enumerate(['室友是','同学是','你是','他是'],1):
    for b,phrase in enumerate(['智障人士服务中心的志愿者','智障人士关爱活动的志愿者'],1):
        add(f'MEDCTX-VOL-{a}-{b}',subject+phrase)
add('MEDCTX-HIGH-1','杀了你','杀了你','HIGH','outside_high_attack')
add('MEDCTX-HIGH-2',"举报：'你个脑摊'请处理。杀了你",'杀了你','HIGH','outside_high_attack')
add('MEDCTX-NARROW-1','你是智障人士','你个智障','MEDIUM','exemption_scope')
add('MEDCTX-NARROW-2','室友是智障人士服务中心的志愿者。你个智障','你个智障','MEDIUM','exemption_scope')
add('MEDCTX-HIGH-3','同学是智障人士关爱活动的志愿者。杀了你','杀了你','HIGH','outside_high_attack')
assert len(cases)==68 and sum(bool(r['expected_word']) for r in cases)==35
data=''.join(json.dumps(row,ensure_ascii=False)+'\n' for row in cases).encode()
path=ROOT/'docs/qa/data/us24-development.v3_8.jsonl'
with path.open('xb') as handle:handle.write(data)
manifest=dict(created_at_utc=datetime.now(UTC).isoformat(),source='M6 synthetic campus development',
    label_scope='literal mapped variant and narrow volunteer-context proposal; not independent semantic gold',
    rows=68,positive=35,negative=33,sha256=hashlib.sha256(data).hexdigest(),
    M1_review='PENDING',independent_acceptance=False,gradient_training=False)
with path.with_suffix('.manifest.json').open('x',encoding='utf-8') as handle:json.dump(manifest,handle,ensure_ascii=False,indent=2)
print(json.dumps(manifest))
