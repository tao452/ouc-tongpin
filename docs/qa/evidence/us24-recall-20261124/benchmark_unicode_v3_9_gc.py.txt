events=[]
active={}
started_collections={}
def track_gc(phase, info):
    generation=info['generation']
    if phase=='start':started_collections[generation]=time.perf_counter()
    else:events.append(dict(group=active.get('group'),iteration=active.get('iteration'),generation=generation,
        duration_ms=(time.perf_counter()-started_collections.pop(generation))*1000,collected=info['collected']))
gc.callbacks.append(track_gc)
measurements = {}
for key, unit in units.items():
    text = (unit * (2000 // len(unit) + 1))[:2000]
    timings, errors = [], []
    for index in range(100):
        active.update(group=key,iteration=index)
        started = time.perf_counter()
        try:
            service.check(text)

        timings.append((time.perf_counter() - started) * 1000)
