def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()

def score(rows):
    report={}
    for mode in ['any_level','high_blocking']:
        cm=dict(tp=0,fp=0,fn=0,tn=0)
        for row in rows:
            gold=bool(row['expected_hits']) if mode=='any_level' else row['blocked']
            pred=(not gold if row['error'] else bool(row['response']['hits'])
                  if mode=='any_level' else not row['response']['passed'])
            cm['tp' if pred and gold else 'fp' if pred else 'fn' if gold else 'tn']+=1
        tp,fp,fn,tn=(cm[k] for k in ['tp','fp','fn','tn'])
        report[mode]=dict(confusion_matrix=cm,recall=tp/(tp+fn) if tp+fn else None,
                         precision=tp/(tp+fp) if tp+fp else None,fpr=fp/(fp+tn) if fp+tn else None)
    return report
