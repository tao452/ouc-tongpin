ROOT=Path.cwd();LAB=ROOT/'qa-private/us24-recall-20261124'
source_sha=lambda p:hashlib.sha256(p.read_bytes().replace(b'\r\n',b'\n')).hexdigest()


def digest(path):
    with path.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()


def create_model():
    directory=LAB/'rocbert-base'
    manifest=json.loads((LAB/'rocbert-base.manifest.json').read_text(encoding='utf-8'))
    assert manifest['state']=='ALL_SELECTED_ASSETS_VERIFIED'
    assert digest(directory/'pytorch_model.bin')==manifest['weights']['sha256']
    for entry in manifest['metadata_assets']:assert digest(directory/entry['file'])==entry['sha256']
    config=AutoConfig.from_pretrained(str(directory),local_files_only=True,trust_remote_code=False)
    assert config.model_type=='roc_bert' and config.num_hidden_layers==12
    config.num_labels=2;config.id2label={0:'nonoffensive',1:'offensive'}
    config.label2id={'nonoffensive':0,'offensive':1};config.use_cache=False
    model,loading=AutoModelForSequenceClassification.from_pretrained(str(directory),config=config,
        local_files_only=True,trust_remote_code=False,weights_only=True,attn_implementation='eager',output_loading_info=True)
    assert all(k.startswith(('classifier.','roc_bert.pooler.')) for k in loading['missing_keys']),loading
    assert all(k.startswith('cls.') or k.endswith('.position_ids') for k in loading['unexpected_keys']),loading
    assert not loading['mismatched_keys'] and not loading['error_msgs']
    return model.cuda(),dict(base_manifest=manifest,loading=loading)


def inputs(windows):
    longest=max(len(w['input_ids']) for w in windows);assert longest<=512
    batch={k:torch.zeros((len(windows),longest),dtype=torch.long) for k in
        ['input_ids','input_shape_ids','input_pronunciation_ids','attention_mask']}
    for i,window in enumerate(windows):
        size=len(window['input_ids'])
        for key in ['input_ids','input_shape_ids','input_pronunciation_ids']:
            assert len(window[key])==size;batch[key][i,:size]=torch.tensor(window[key],dtype=torch.long)
        batch['attention_mask'][i,:size]=1
    return {k:v.cuda() for k,v in batch.items()}


def save(model,path):
    save_file({name:value.detach().cpu().contiguous() for name,value in model.state_dict().items()},str(path))


def predict(model,records):
    model.eval();flat=[(i,w) for i,r in enumerate(records) for w in r['windows']]
    flat.sort(key=lambda x:len(x[1]['input_ids']))
    scores=[None]*len(records);errors={};began=time.perf_counter()
    with torch.inference_mode():
        for start in range(0,len(flat),16):
            batch=flat[start:start+16]
            try:
                with torch.autocast('cuda',dtype=torch.bfloat16):logits=model(**inputs([w for _,w in batch])).logits
                probabilities=logits.float().softmax(dim=-1)[:,1].cpu().tolist()
                for (index,_),score in zip(batch,probabilities,strict=True):
                    if not math.isfinite(score):errors[index]='NonFinite'
                    else:scores[index]=max(scores[index] or 0,score)
            except RuntimeError as error:
                for index,_ in batch:errors[index]=type(error).__name__
                torch.cuda.empty_cache()
    rows=[]
    for i,record in enumerate(records):
        row={k:v for k,v in record.items() if k!='windows'}
        if i in errors or scores[i] is None:row['error']=errors.get(i,'MissingScore')
        else:row['score']=scores[i]
        rows.append(row)
    return rows,time.perf_counter()-began


def main(probe):
    assert torch.cuda.is_available() and torch.cuda.is_bf16_supported()
    torch.set_num_threads(4);random.seed(20261124);np.random.seed(20261124)
    torch.manual_seed(20261124);torch.cuda.manual_seed_all(20261124)
    torch.backends.cudnn.deterministic=True;torch.backends.cudnn.benchmark=False
    torch.use_deterministic_algorithms(True,warn_only=True)
    path=LAB/'rocbert-v8-inputs/features.pt';prepared=torch.load(path,map_location='cpu',weights_only=True)
    policy,records=prepared['policy'],prepared['records']
    for name,value in policy['evaluation_policy']['current_engine_sha256'].items():assert source_sha(ROOT/name)==value
    for name,value in policy['code_sha256'].items():assert source_sha(LAB/name)==value
    training=[(w,r['label'],r['id'],i) for r in records if r['cohort']=='training' for i,w in enumerate(r['windows'])]
    evaluation=[r for r in records if r['cohort']!='training']
    assert len(evaluation)==3258 and len(training)==15727
    model,base=create_model();trainable=[p for p in model.parameters() if p.requires_grad]
    assert all(p.dtype==torch.float32 for p in trainable)
    optimizer=torch.optim.AdamW(trainable,lr=2e-5,weight_decay=.01,foreach=False)
    code={p.name:source_sha(p) for p in [Path(__file__),LAB/'rocbert_encoding_v8.py',LAB/'phonetic_recheck_metrics.py']}
    metadata=dict(created_at_utc=datetime.now(UTC).isoformat(),dataset_policy=policy,base=base,
        features_sha256=digest(path),index_sha256=digest(path.with_name('index.json')),code_sha256=code,
        trainable_parameters=sum(p.numel() for p in trainable),epochs=3,seed=20261124,
        optimizer='AdamW foreach=False',learning_rate=2e-5,weight_decay=.01,gradient_clip=1.,
        schedule='5% total3epoch optimizer-step warmup then constant',training_rows=15698,
        training_windows_per_epoch=len(training),evaluation_rows=3258,excluded_evaluation_rows=0,
        training_time_budget_minutes=180,prediction_microbatch=16,
        runtime=dict(torch=torch.__version__,cuda=torch.version.cuda,device=torch.cuda.get_device_name(0),
            packages={p:importlib.metadata.version(p) for p in ['transformers','tokenizers','safetensors','pypinyin','numpy']}),
        production_acceptance=False)
    if probe:
        assert probe in {4,8}
        worst=sorted(training,key=lambda x:len(x[0]['input_ids']),reverse=True)[:probe]
        model.train();torch.cuda.reset_peak_memory_stats();began=time.perf_counter()
        for stage in range(3):
            with torch.autocast('cuda',dtype=torch.bfloat16):
                logits=model(**inputs([w for w,_,_,_ in worst])).logits.float()
                loss=F.cross_entropy(logits,torch.tensor([y for _,y,_,_ in worst],device='cuda'))
            assert logits.shape==(probe,2) and torch.isfinite(loss)
            loss.backward()
            if stage in {0,2}:
                assert all(p.grad is not None and torch.isfinite(p.grad).all() for p in trainable)
                torch.nn.utils.clip_grad_norm_(trainable,1.);optimizer.step();optimizer.zero_grad(set_to_none=True)
        metadata.update(micro_batch_size=probe,accumulate_steps=32//probe,effective_batch=32,
            probe_microbatches=3,probe_optimizer_updates=2,probe_updates_reused=False,accuracy_evaluated=False,
            peak_cuda_bytes=torch.cuda.max_memory_allocated(),physical_total_bytes=torch.cuda.get_device_properties(0).total_memory,
            elapsed_seconds=time.perf_counter()-began)
        metadata['physical_budget_pass']=metadata['peak_cuda_bytes']<.92*metadata['physical_total_bytes']
        with (LAB/f'rocbert-v8-preflight-{probe}.json').open('x',encoding='utf-8') as f:json.dump(metadata,f,ensure_ascii=False,indent=2)
        print(json.dumps({k:metadata[k] for k in ['physical_budget_pass','peak_cuda_bytes','elapsed_seconds','trainable_parameters']}),flush=True)
        return 0 if metadata['physical_budget_pass'] else 2
    choices=[]
    for size in [8,4]:
        check=LAB/f'rocbert-v8-preflight-{size}.json'
        if check.exists():
            value=json.loads(check.read_text(encoding='utf-8'))
            assert value['features_sha256']==metadata['features_sha256'] and value['code_sha256']==code
            if value['physical_budget_pass']:choices.append(value)
    assert choices
    selected=choices[0];micro=selected['micro_batch_size'];accum=32//micro
    metadata.update(micro_batch_size=micro,accumulate_steps=accum,effective_batch=32,
        preflight_sha256=digest(LAB/f'rocbert-v8-preflight-{micro}.json'),probe_updates_reused=False)
    output=LAB/'rocbert-v8';output.mkdir(exist_ok=False);orders=[]
    for epoch in [1,2,3]:
        rng=random.Random(20261124+epoch);order=list(range(len(training)));rng.shuffle(order);batches=[]
        for start in range(0,len(order),256):
            pool=sorted(order[start:start+256],key=lambda i:len(training[i][0]['input_ids']))
            batches.extend(pool[i:i+micro] for i in range(0,len(pool),micro))
        rng.shuffle(batches);assert sorted(i for b in batches for i in b)==list(range(len(training)))
        orders.append(batches)
    with (output/'training-window-order.json').open('x',encoding='utf-8') as f:json.dump(orders,f)
    metadata['training_order_sha256']=digest(output/'training-window-order.json')
    with (output/'frozen.json').open('x',encoding='utf-8') as f:json.dump(metadata,f,ensure_ascii=False,indent=2)
    update=0;results=[];deadline=time.monotonic()+180*60
    warmup=max(1,math.ceil(sum(math.ceil(len(b)/accum) for b in orders)*.05))
    for epoch,batches in enumerate(orders,1):
        model.train();optimizer.zero_grad(set_to_none=True);began=time.perf_counter();seen=0;total_loss=0.
        for step,indices in enumerate(batches,1):
            if time.monotonic()>deadline:raise TimeoutError('RoCBERT exceeded180-minute training budget')
            batch=[training[i] for i in indices];group_start=((step-1)//accum)*accum
            group_count=sum(len(b) for b in batches[group_start:group_start+accum])
            with torch.autocast('cuda',dtype=torch.bfloat16):
                logits=model(**inputs([w for w,_,_,_ in batch])).logits.float()
                loss=F.cross_entropy(logits,torch.tensor([y for _,y,_,_ in batch],device='cuda'))
            if not torch.isfinite(loss):raise RuntimeError('Nonfinite robustencoder loss')
            total_loss+=float(loss.detach())*len(batch);seen+=len(batch)
            (loss*len(batch)/group_count).backward()
            if step%accum==0 or step==len(batches):
                update+=1
                for group in optimizer.param_groups:group['lr']=2e-5*min(1.,update/warmup)
                torch.nn.utils.clip_grad_norm_(trainable,1.);optimizer.step();optimizer.zero_grad(set_to_none=True)
                if update%250==0:
                    partial=output/f'partial-epoch-{epoch}-step-{step}';partial.mkdir(exist_ok=False)
                    save(model,partial/'model.safetensors')
                    torch.save(dict(optimizer=optimizer.state_dict(),torch_rng=torch.get_rng_state(),cuda_rng=torch.cuda.get_rng_state_all()),partial/'optimizer-and-rng.pt')
                    with (partial/'progress.json').open('x',encoding='utf-8') as f:json.dump(dict(epoch=epoch,batches=step,total_batches=len(batches),windows_seen=seen,epoch_complete=False,accuracy_status='NOT_RUN',evaluation_run=False,features_sha256=metadata['features_sha256'],code_sha256=code),f,indent=2)
            if step%100==0:print(f'RoCBERT epoch={epoch} batch={step}/{len(batches)} loss={total_loss/seen:.5f} elapsed={time.perf_counter()-began:.1f}s',flush=True)
        assert seen==len(training);seconds=time.perf_counter()-began
        directory=output/f'epoch-{epoch}';directory.mkdir(exist_ok=False);weights=directory/'model.safetensors';save(model,weights)
        predictions,inference_seconds=predict(model,evaluation)
        with (directory/'predictions.jsonl').open('x',encoding='utf-8') as f:
            for row in predictions:f.write(json.dumps(row,ensure_ascii=False)+'\n')
        result=dict(epoch=epoch,weights_sha256=digest(weights),training_seconds=seconds,inference_seconds=inference_seconds,
            training_windows_seen=seen,training_loss=total_loss/seen,evaluation_rows=len(predictions),excluded_evaluation_rows=0,
            peak_cuda_bytes=torch.cuda.max_memory_allocated(),metrics=summarize(predictions,directory/'thresholds.csv'),production_acceptance=False)
        with (directory/'report.json').open('x',encoding='utf-8') as f:json.dump(result,f,ensure_ascii=False,indent=2)
        results.append(result)
        with (output/f'summary-through-epoch-{epoch}.json').open('x',encoding='utf-8') as f:json.dump(dict(frozen=metadata,epochs=results),f,ensure_ascii=False,indent=2)
        print(json.dumps(dict(epoch=epoch,selected=result['metrics']['literal_or_model']['best_fpr_eligible'])),flush=True)
    with (output/'summary.json').open('x',encoding='utf-8') as f:json.dump(dict(frozen=metadata,epochs=results),f,ensure_ascii=False,indent=2)
    return 0 if any(m['best_fpr_eligible'] and m['best_fpr_eligible']['development_gate_met'] for r in results for m in r['metrics'].values()) else 1


if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--probe-batch',type=int,default=0)
    raise SystemExit(main(p.parse_args().probe_batch))
