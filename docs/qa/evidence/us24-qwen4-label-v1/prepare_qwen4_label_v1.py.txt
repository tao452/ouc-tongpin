import hashlib
import json
import sys
from collections import Counter
from datetime import UTC, datetime
from pathlib import Path

import torch

from qwen4_label_encoding_v1 import LabelEncoder, SYSTEM

ROOT = Path.cwd()
LAB = ROOT / 'qa-private/us24-recall-20261124'
sys.path.insert(0, str(ROOT/'services/algo/scripts'))
from evaluate_candidate import development, pcr
from evaluate_recall import load_external, source_sha
from app.services.normalization import canonicalize


def sha(path):
    with path.open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()


def main():
    torch.set_num_threads(2)
    assets = json.loads((LAB/'qwen4-instruct-assets-preparation.json').read_text(encoding='utf-8'))
    directory = LAB/'qwen3-4b-instruct2507'
    for entry in assets['metadata_assets']:
        assert sha(directory/entry['file']) == entry['sha256']
    previous_path = LAB/'rocbert-v9-inputs/features.pt'
    previous = torch.load(previous_path, map_location='cpu', weights_only=True)
    evaluation = [r for r in previous['records'] if r['cohort'] != 'training']
    assert len(evaluation) == 3258
    raw = {r['id']:r['text'] for r in load_external(LAB/'toxcn.csv', json.loads(
        (ROOT/'docs/qa/data/external/toxicn-v3.sealed.json').read_text(encoding='utf-8')))}
    pcr_rows, excluded = pcr(ROOT/'qa-private/us24-m1/PCR-ToxiCN.json', json.loads(
        (ROOT/'docs/qa/data/external/pcr-test.v1.json').read_text(encoding='utf-8')))
    assert not excluded
    raw.update({r['id']:r['text'] for r in pcr_rows})
    for version in ['v2','v3']:
        for row in development(ROOT/f'docs/qa/data/us24-development.{version}.jsonl'):
            if not row['expected_word']:
                raw[f'{version}-{row["id"]}'] = row['text']
    for row in map(json.loads, (ROOT/'docs/qa/data/us24-context-stress.v1.jsonl').read_text(encoding='utf-8').splitlines()):
        raw[row['id']] = row['text']
    encoder = LabelEncoder(directory)
    output = LAB/'qwen4-label-v1-inputs'
    output.mkdir(exist_ok=False)
    ready = []
    index = []
    changes = Counter()
    for number, row in enumerate(evaluation, 1):
        source = raw[row['id']]
        assert hashlib.sha256(source.encode()).hexdigest() == row['source_text_sha256']
        text = canonicalize(source)
        windows = encoder.encode(text)
        metadata = {k:v for k,v in row.items() if k not in {'windows','representation_sha256'}}
        metadata.update(previous_stage_transformed_sha256=row['transformed_sha256'],
                        transformed_sha256=hashlib.sha256(text.encode()).hexdigest())
        changes[row['cohort']] += metadata['transformed_sha256'] != row['transformed_sha256']
        ready.append({**metadata, 'windows':windows})
        index.append({**metadata, 'windows':len(windows), 'window_lengths':[len(w['input_ids']) for w in windows],
                      'all_source_characters_covered':True})
        if number % 500 == 0:
            print(f'Prepared fixed label-token instruction inputs {number}/3258', flush=True)
    assert len(ready) == len({r['id'] for r in ready}) == 3258
    policy = dict(created_at_utc=datetime.now(UTC).isoformat(), scope='fixed zero-gradient 4B instruction-model development',
                  assets=assets, previous_features_sha256=sha(previous_path), prompt=SYSTEM,
                  prompt_sha256=encoder.prompt_sha256, label_token_ids=encoder.label_tokens,
                  label_semantics={'0':'no broad offensive intent', '1':'broad offensive intent; not S1 severity'},
                  input='canonical complete raw comment plus one context-first toneless reading; English runs retained',
                  input_policy='same global prompt and preprocessing for all rows; no corpus/category routing',
                  maximum_tokens=2048, character_overlap=32, maximum_windows=32,
                  window_policy='all source characters covered; maximum offensive score; any window error counts worst for row',
                  malformed_output_policy='unconstrained top next token must be0or1; otherwise error, not a guessed label',
                  score='softmax restricted to the two explicit digit logits; record total digit mass and actual top token',
                  original_rows=3258, excluded_rows=0, gradient_updates=0,
                  changed_input_text_by_cohort=dict(changes),
                  labels_and_literal_flags='same original3258 and SHA-verified current3.10 cached literal flags',
                  dictionary_sha256='4dbc9fed5baab58f0e11f2411eec761d7f01d5f44080cbbaa0e7e7b8ee0c2722',
                  code_sha256={p.name:source_sha(p) for p in [Path(__file__),LAB/'qwen4_label_encoding_v1.py']},
                  upstream_pretraining_evaluation_exposure='unknown', COLD_dev_fetched=False,
                  M1_review='PENDING', M3_reproduction='PENDING', production_acceptance=False)
    with (output/'features.pt').open('xb') as handle:
        torch.save(dict(policy=policy, records=ready), handle)
    for name, value in [('policy.json',policy),('index.json',dict(policy=policy,rows=index))]:
        with (output/name).open('x',encoding='utf-8') as handle:
            json.dump(value,handle,ensure_ascii=False,indent=2)
    print(json.dumps(dict(rows=3258,windows=sum(len(r['windows']) for r in ready),
                          label_tokens=encoder.label_tokens,changed_texts=dict(changes))),flush=True)


if __name__ == '__main__':
    main()
