import hashlib
import json
import math
from datetime import UTC, datetime
from pathlib import Path

from phonetic_recheck_metrics import summarize

ROOT=Path.cwd();LAB=ROOT/'qa-private/us24-recall-20261124'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
paths=[LAB/'phonetic-training-v1-context36/epoch-3/predictions.jsonl',
       LAB/'phonetic-training-v2/epoch-2/predictions.jsonl',
       LAB/'phonetic-training-v3/epoch-1/predictions.jsonl',
       LAB/'linear-phonetic-v1/predictions.jsonl']
sources=[]
for path in paths:
    rows=[json.loads(line) for line in path.read_text(encoding='utf-8').splitlines()]
    assert len(rows)==3258 and len({row['id'] for row in rows})==3258
    sources.append({row['id']:row for row in rows})
keys=list(sources[0])
for source in sources[1:]:
    assert source.keys()==sources[0].keys()
    for identifier in keys:
        for key in ['label','cohort','category','source_text_sha256','normalized_sha256','transformed_sha256','literal_any']:
            assert source[identifier][key]==sources[0][identifier][key]
output=LAB/'phonetic-consensus-v1';output.mkdir(exist_ok=False)
frozen=dict(created_at_utc=datetime.now(UTC).isoformat(),scope='fixed development consensus; no retraining or new independent evaluation',
    code_sha256=hashlib.sha256(Path(__file__).read_bytes().replace(b'\r\n',b'\n')).hexdigest(),
    metric_code_sha256=hashlib.sha256((LAB/'phonetic_recheck_metrics.py').read_bytes().replace(b'\r\n',b'\n')).hexdigest(),
    sources=[dict(path=p.relative_to(LAB).as_posix(),sha256=sha(p)) for p in paths],
    fixed_checkpoints='v1 epoch3, v2 epoch2, v3 epoch1, linear-v1; no per-row or category switching',
    methods=['equal_probability_mean','equal_logit_mean','maximum_score_diagnostic'],
    score_note='uncalibrated model scores; development thresholds selected with same four FPR/Recall gates',
    evaluation_rows=3258,excluded_evaluation_rows=0,production_acceptance=False)
with (output/'frozen.json').open('x',encoding='utf-8') as handle:json.dump(frozen,handle,ensure_ascii=False,indent=2)
results={}
for method in frozen['methods']:
    directory=output/method;directory.mkdir(exist_ok=False);predictions=[]
    for identifier in keys:
        rows=[source[identifier] for source in sources]
        row={k:v for k,v in rows[0].items() if k not in {'score','windows','window_lengths','error'}}
        if any(r.get('error') or not math.isfinite(r.get('score',float('nan'))) for r in rows):
            row['error']='AnySourceInferenceError'
        else:
            scores=[r['score'] for r in rows]
            if method=='equal_probability_mean':score=sum(scores)/len(scores)
            elif method=='maximum_score_diagnostic':score=max(scores)
            else:
                clipped=[min(1-1e-6,max(1e-6,score)) for score in scores]
                average=sum(math.log(p/(1-p)) for p in clipped)/len(clipped)
                score=1/(1+math.exp(-average))
            row['score']=score
        predictions.append(row)
    with (directory/'predictions.jsonl').open('x',encoding='utf-8') as handle:
        for row in predictions:handle.write(json.dumps(row,ensure_ascii=False)+'\n')
    result=dict(method=method,evaluation_rows=len(predictions),excluded_evaluation_rows=0,
        metrics=summarize(predictions,directory/'thresholds.csv'),production_acceptance=False)
    with (directory/'report.json').open('x',encoding='utf-8') as handle:json.dump(result,handle,ensure_ascii=False,indent=2)
    results[method]=result
    point=result['metrics']['literal_or_model']['best_fpr_eligible']
    print(json.dumps(dict(method=method,best=point)),flush=True)
with (output/'summary.json').open('x',encoding='utf-8') as handle:json.dump(dict(frozen=frozen,results=results),handle,ensure_ascii=False,indent=2)
raise SystemExit(0 if any(p and p['development_gate_met'] for r in results.values() for m in r['metrics'].values() for p in [m['best_fpr_eligible']]) else 1)
