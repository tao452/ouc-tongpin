import time
from datetime import UTC, datetime
from pathlib import Path

import numpy as np
import torch
from safetensors.torch import save_file

from chinesebert_runtime.modeling_glycebert import GlyceBertForSequenceClassification

ROOT=Path.cwd();LAB=ROOT/'qa-private/us24-recall-20261124'
sha=lambda path:hashlib.sha256(path.read_bytes()).hexdigest()
source_sha=lambda path:hashlib.sha256(path.read_bytes().replace(b'\r\n',b'\n')).hexdigest()


def batches(items,size,shuffle=False):
    indices=list(range(len(items)))
    if shuffle:random.shuffle(indices)
    for start in range(0,len(indices),size):yield [items[i] for i in indices[start:start+size]]


def inputs(items):
    longest=max(len(window['input_ids']) for window in items)
    tokens=torch.zeros((len(items),longest),dtype=torch.long)
    phonetics=torch.zeros((len(items),longest,8),dtype=torch.long)
    mask=torch.zeros_like(tokens)
    for index,window in enumerate(items):
        length=len(window['input_ids'])
        tokens[index,:length]=torch.tensor(window['input_ids'],dtype=torch.long)
        phonetics[index,:length]=torch.tensor(window['pinyin_ids'],dtype=torch.long)
        mask[index,:length]=1
    return dict(input_ids=tokens.cuda(),pinyin_ids=phonetics.cuda(),attention_mask=mask.cuda())


def create_model():
    model_dir=LAB/'chinesebert-model'
    manifest=json.loads((LAB/'chinesebert-model.manifest.json').read_text(encoding='utf-8'))
    assert sha(model_dir/'pytorch_model.bin')==manifest['files']['pytorch_model.bin']['sha256']
    config=BertConfig.from_json_file(str(model_dir/'config.json'))
    config.name_or_path=str(model_dir);config.num_labels=2;config._attn_implementation='eager'
    config.id2label={0:'nonoffensive',1:'offensive'};config.label2id={'nonoffensive':0,'offensive':1}
    model=GlyceBertForSequenceClassification(config)
    state=torch.load(model_dir/'pytorch_model.bin',map_location='cpu',weights_only=True)
    loaded=model.load_state_dict(state,strict=False)
    assert all(k.startswith('classifier.') or k.endswith('.position_ids') for k in loaded.missing_keys)
    assert all(k.startswith('cls.') for k in loaded.unexpected_keys)
    assert 'bert.embeddings.glyph_embeddings.embedding.weight' in state
    model.bert.embeddings.glyph_embeddings.embedding.weight.requires_grad_(False)
    return model.cuda()


def predict(model,records):
    model.eval()
    flattened=[(index,window) for index,row in enumerate(records) for window in row['windows']]
    scores=[None]*len(records);errors={}
    began=time.perf_counter()
    with torch.inference_mode():
        for batch in batches(flattened,16):
            try:
                with torch.autocast('cuda',dtype=torch.bfloat16):
                    logits=model(**inputs([w for _,w in batch])).logits
                probabilities=logits.float().softmax(dim=-1)[:,1].cpu().tolist()
                for (index,_),score in zip(batch,probabilities,strict=True):
                    if not math.isfinite(score):errors[index]='NonFinite'
                    else:scores[index]=max(scores[index] or 0,score)
            except RuntimeError as error:
                for index,_ in batch:errors[index]=type(error).__name__
                torch.cuda.empty_cache()
    result=[]
    for index,row in enumerate(records):
        item={k:v for k,v in row.items() if k!='windows'}
        if index in errors or scores[index] is None:item['error']=errors.get(index,'MissingScore')
        else:item['score']=scores[index]
        result.append(item)
    return result,time.perf_counter()-began


def main():
    if not torch.cuda.is_available() or not torch.cuda.is_bf16_supported():
        raise RuntimeError('Verified CUDA/BF16 runtime is required')
    torch.set_num_threads(4);random.seed(20261124);np.random.seed(20261124)
    torch.manual_seed(20261124);torch.cuda.manual_seed_all(20261124)
    torch.backends.cudnn.deterministic=True;torch.backends.cudnn.benchmark=False
    torch.use_deterministic_algorithms(True,warn_only=True)
    features=LAB/'phonetic-training-v1.features.pt'
    prepared=torch.load(features,map_location='cpu',weights_only=True)
    policy=prepared['policy'];records=prepared['records']
    assert len(records)==12768
    for name,digest in policy['engine_sha256'].items():assert source_sha(ROOT/name)==digest
    training=[(window,row['label']) for row in records if row['cohort']=='training' for window in row['windows']]
    evaluation=[row for row in records if row['cohort'] in {'toxicn_dev','PCR','controls'}]
    assert sum(row['cohort']=='training' for row in records)==9612 and len(evaluation)==3156
    output=LAB/'phonetic-training-v1';output.mkdir(exist_ok=False)
    runtime=dict(torch=torch.__version__,cuda=torch.version.cuda,device=torch.cuda.get_device_name(0),
                 packages={p:importlib.metadata.version(p) for p in ['transformers','tokenizers','pypinyin','safetensors']})
    frozen=dict(dataset_policy=policy,features_sha256=sha(features),
        code_sha256={p.name:source_sha(p) for p in [Path(__file__),LAB/'phonetic_training_metrics.py',LAB/'phonetic_encoding.py']},
        native_code_sha256={p.name:source_sha(p) for p in (LAB/'chinesebert-code/chinesebert_runtime').glob('*.py')},
        pretrained_sha256=json.loads((LAB/'chinesebert-model.manifest.json').read_text(encoding='utf-8')),
        optimizer='AdamW',weight_decay=.01,gradient_clip=1.0,learning_rate_schedule='constant',precision='bfloat16',
        selection='maximize minimum PCR and ToxiCN development recall subject to all3 FPR<2%; zero errors',
        training_rows=9612,training_windows=len(training),evaluation_rows=len(evaluation),runtime=runtime,
        production_enabled=False)
    with (output/'frozen.json').open('x',encoding='utf-8') as handle:json.dump(frozen,handle,ensure_ascii=False,indent=2)
    model=create_model();optimizer=torch.optim.AdamW([p for p in model.parameters() if p.requires_grad],lr=2e-5,weight_decay=.01)
    deadline=time.monotonic()+45*60
    torch.cuda.reset_peak_memory_stats();all_results=[]
    for epoch in range(1,4):
        directory=output/f'epoch-{epoch}';directory.mkdir(exist_ok=False)
        model.train();optimizer.zero_grad(set_to_none=True);loss_sum=0;seen=0;began=time.perf_counter()
        batch_count=math.ceil(len(training)/8)
        for step,batch in enumerate(batches(training,8,True),1):
            if time.monotonic()>deadline:raise TimeoutError('Training exceeded45-minute experiment budget')
            with torch.autocast('cuda',dtype=torch.bfloat16):
                labels=torch.tensor([label for _,label in batch],device='cuda',dtype=torch.long)
                loss=model(**inputs([window for window,_ in batch]),labels=labels).loss
            if not torch.isfinite(loss):raise RuntimeError('Training loss is not finite')
            loss_sum+=float(loss.detach())*len(batch);seen+=len(batch)
            accumulation=min(4,batch_count-((step-1)//4)*4)
            (loss/accumulation).backward()
            if step%4==0 or step==batch_count:
                torch.nn.utils.clip_grad_norm_([p for p in model.parameters() if p.requires_grad],1.0)
                optimizer.step();optimizer.zero_grad(set_to_none=True)
            if step%100==0:print(f'epoch={epoch} batch={step}/{batch_count} loss={loss_sum/seen:.5f}',flush=True)
        train_seconds=time.perf_counter()-began
        weights=directory/'model.safetensors'
        save_file({k:v.detach().cpu().contiguous() for k,v in model.state_dict().items()},str(weights))
        model.config.to_json_file(str(directory/'config.json'))
        predictions,inference_seconds=predict(model,evaluation)
        with (directory/'predictions.jsonl').open('x',encoding='utf-8') as handle:
            for row in predictions:handle.write(json.dumps(row,ensure_ascii=False)+'\n')
        result=dict(epoch=epoch,executed_at_utc=datetime.now(UTC).isoformat(),weights_sha256=sha(weights),
            training_loss=loss_sum/seen,training_seconds=train_seconds,inference_seconds=inference_seconds,
            peak_cuda_bytes=torch.cuda.max_memory_allocated(),evaluation_rows=len(predictions),excluded_evaluation_rows=0,
            metrics=summarize(predictions,directory/'thresholds.csv'),production_acceptance=False)
        with (directory/'report.json').open('x',encoding='utf-8') as handle:json.dump(result,handle,ensure_ascii=False,indent=2)
        all_results.append(result)
        print(json.dumps(dict(epoch=epoch,training_loss=result['training_loss'],
            model=result['metrics']['model']['best_fpr_eligible'],ensemble=result['metrics']['literal_or_model']['best_fpr_eligible'])),flush=True)
    with (output/'summary.json').open('x',encoding='utf-8') as handle:json.dump(dict(frozen=frozen,epochs=all_results),handle,ensure_ascii=False,indent=2)
    return 0 if any(point and point['development_gate_met'] for result in all_results for value in result['metrics'].values() for point in [value['best_fpr_eligible']]) else 1


if __name__=='__main__':raise SystemExit(main())
