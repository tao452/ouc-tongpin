import csv
import hashlib
import json
import sys
from collections import Counter, defaultdict
from datetime import UTC, datetime
from pathlib import Path

import torch
from pypinyin import Style, pinyin

from qwen_sequence_encoding_v5 import SequenceEncoder

ROOT = Path.cwd()
LAB = ROOT / 'qa-private/us24-recall-20261124'
sys.path.insert(0, str(ROOT / 'services/algo/scripts'))
from evaluate_recall import load_external, source_sha
from app.services.normalization import canonicalize


def sha(text):
    return hashlib.sha256(text.encode()).hexdigest()


def readings(text):
    result = [part[0] for part in pinyin(text, style=Style.NORMAL, heteronym=False,
                                       errors=lambda value: list(value))]
    assert len(result) == len(text)
    return result


def main():
    torch.set_num_threads(2)
    previous = LAB / 'qwen-sequence-v5-inputs/features.pt'
    prepared = torch.load(previous, map_location='cpu', weights_only=True)
    original = [r for r in prepared['records'] if r['cohort'] == 'training' and not r.get('augmentation')]
    training = [r for r in prepared['records'] if r['cohort'] == 'training']
    assert len(original) == 35293 and len(training) == 38856
    current_file = LAB / 'qwen-sequence-v5-context39/features.pt'
    current = torch.load(current_file, map_location='cpu', weights_only=True)
    evaluation = current['records']
    assert len(evaluation) == 3258
    for name, value in current['policy']['current_engine_sha256'].items():
        assert source_sha(ROOT / name) == value
    raw = {r['id']: r['text'] for r in load_external(LAB / 'toxcn.csv', json.loads(
        (ROOT / 'docs/qa/data/external/toxicn-v3.sealed.json').read_text(encoding='utf-8')))}
    cold = LAB / 'cold-official-train.csv'
    assert hashlib.sha256(cold.read_bytes()).hexdigest() == '16ca7955bdcaf90f5d8ac39e51eb3eda15320caa1e730ed45cb5e8ca8cd0d5e8'
    with cold.open(encoding='utf-8-sig', newline='') as handle:
        for number, row in enumerate(csv.DictReader(handle), 1):
            raw[f'COLD-TRAIN-{number:05d}'] = row['TEXT']
    output = LAB / 'qwen-homophone-v6-inputs'
    output.mkdir(exist_ok=False)
    frequency = defaultdict(Counter)
    views = {}
    for number, row in enumerate(original, 1):
        source = raw[row['id']]
        assert sha(source) == row['source_text_sha256']
        text = canonicalize(source)
        assert sha(text) == row['transformed_sha256']
        sounds = readings(text)
        views[row['id']] = (text, sounds)
        for char, sound in zip(text, sounds, strict=True):
            if '\u4e00' <= char <= '\u9fff' and sound.isascii() and sound.isalpha():
                frequency[sound][char] += 1
        if number % 5000 == 0:
            print(f'Whole-context training readings {number}/35293', flush=True)
    # Only training text determines the common-character table.
    table = {sound: [c for c, count in sorted(counts.items(), key=lambda item: (-item[1], item[0])) if count >= 20]
             for sound, counts in frequency.items()}
    evaluation_hashes = {r[k] for r in evaluation for k in ['normalized_sha256', 'transformed_sha256']}
    encoder = SequenceEncoder(LAB / 'qwen3-1_7b-tokenizer')
    added, audit = [], []
    for row in original:
        if int(sha('us24-homophone-v6:' + row['source_text_sha256']), 16) % 10:
            continue
        text, sounds = views[row['id']]
        positions = [i for i, char in enumerate(text) if '\u4e00' <= char <= '\u9fff'
                     and any(c != char for c in table.get(sounds[i], []))]
        positions.sort(key=lambda i: sha(row['id'] + ':' + str(i)))
        attempts, accepted = [], None
        for index in positions[:2]:
            candidates = [c for c in table.get(sounds[index], []) if c != text[index]][:3]
            for char in candidates:
                value = text[:index] + char + text[index + 1:]
                unchanged = readings(value) == sounds
                overlap = sha(value) in evaluation_hashes
                attempts.append(dict(position=index, replacement_sha256=sha(char),
                    transformed_sha256=sha(value), whole_readings_unchanged=unchanged,
                    evaluation_overlap=overlap))
                if unchanged and not overlap:
                    accepted = value
                    break
            if accepted is not None:
                break
        item = dict(source_id=row['id'], label=row['label'], selected_without_labels=True,
                    source_text_sha256=row['source_text_sha256'], attempts=attempts,
                    accepted=accepted is not None)
        if accepted is not None:
            record = {k: v for k, v in row.items() if k != 'windows'}
            record.update(id='HOM-V6-' + row['id'], source_id=row['id'],
                augmentation='single_common_hanzi_context_homophone_assumed_label',
                previous_transform_sha256=row['transformed_sha256'],
                transformed_sha256=sha(accepted), windows=encoder.encode(accepted))
            assert len(accepted) == len(text) and sum(a != b for a, b in zip(text, accepted, strict=True)) == 1
            added.append(record)
            item['augmentation_id'] = record['id']
        audit.append(item)
        if len(audit) % 500 == 0:
            print(f'Homophone selected={len(audit)} encoded={len(added)}', flush=True)
    records = training + added + evaluation
    assert len({r['id'] for r in records}) == len(records)
    policy = dict(created_at_utc=datetime.now(UTC).isoformat(),
        scope='development training augmentation; inherited labels are assumptions, not new annotated gold',
        previous_features_sha256=hashlib.sha256(previous.read_bytes()).hexdigest(),
        evaluation_features_sha256=hashlib.sha256(current_file.read_bytes()).hexdigest(),
        current_evaluation_policy=current['policy'], training_rows_before=38856,
        training_rows=len(training) + len(added), original_training_rows_retained=35293,
        previous_pinyin_augmentations_retained=3563, homophone_selected=len(audit),
        homophone_added=len(added), homophone_added_by_label=dict(Counter(str(r['label']) for r in added)),
        selection='SHA256(us24-homophone-v6:+source_text_sha256) mod10 ==0; independent of label and model error',
        replacements='training-only contextual readings/count>=20; at most2positions x3candidates; one replacement; whole no-tone reading sequence must remain identical',
        rejected_augmentation_policy='retain original training row; skip only invalid/overlapping new view',
        maximum_candidate_attempts_per_selected_row=6, maximum_new_views_per_source=1,
        evaluation_rows=3258, excluded_evaluation_rows=0, seed=20261124,
        code_sha256={p.name: source_sha(p) for p in [Path(__file__), LAB / 'qwen_sequence_encoding_v5.py']},
        COLD_dev_fetched=False, M1_review='PENDING', M3_reproduction='PENDING', production_acceptance=False)
    with (output / 'features.pt').open('xb') as handle:
        torch.save(dict(policy=policy, records=records), handle)
    for name, value in [('policy.json', policy), ('augmentation-index.json', audit)]:
        with (output / name).open('x', encoding='utf-8') as handle:
            json.dump(value, handle, ensure_ascii=False, indent=2)
    index = [{**{k: v for k, v in r.items() if k != 'windows'},
              'windows':len(r['windows']), 'window_lengths':[len(w['input_ids']) for w in r['windows']]}
             for r in records]
    with (output / 'index.json').open('x', encoding='utf-8') as handle:
        json.dump(dict(policy=policy, rows=index), handle, ensure_ascii=False, indent=2)
    print(json.dumps(dict(training_rows=policy['training_rows'], homophone_added=len(added),
        evaluation_rows=3258, training_windows=sum(len(r['windows']) for r in records if r['cohort']=='training'))), flush=True)


if __name__ == '__main__':
    main()
