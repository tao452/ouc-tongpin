import argparse
import importlib.metadata
import json
import math
import random
import time
from datetime import UTC, datetime
from pathlib import Path

import numpy as np
import torch
from safetensors.torch import load_file
from torch.nn import functional as F

from phonetic_recheck_metrics import summarize
from rocbert_phonetic_view_v9 import PhoneticView, representation
from train_rocbert_v8 import create_model, digest, inputs, predict, save, source_sha

ROOT = Path.cwd()
LAB = ROOT / 'qa-private/us24-recall-20261124'


def main(probe):
    assert torch.cuda.is_available() and torch.cuda.is_bf16_supported()
    torch.set_num_threads(4)
    random.seed(20261124)
    np.random.seed(20261124)
    torch.manual_seed(20261124)
    torch.cuda.manual_seed_all(20261124)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False
    torch.use_deterministic_algorithms(True, warn_only=True)
    parent = LAB / 'rocbert-v8'
    parent_summary = json.loads((parent / 'summary.json').read_text(encoding='utf-8'))
    assert len(parent_summary['epochs']) == 3 and parent_summary['epochs'][-1]['epoch'] == 3
    for epoch in [1, 2, 3]:
        for filename in ['threshold-grid-gate-audit.json', 'prediction-curve-recomputation.json']:
            audit = json.loads((parent / f'epoch-{epoch}' / filename).read_text(encoding='utf-8'))
            assert not audit['development_gate_met'], 'V8 passed; do not launch failed-gate follow-up'
    parent_weights = parent / 'epoch-3/model.safetensors'
    parent_sha = digest(parent_weights)
    assert parent_sha == parent_summary['epochs'][-1]['weights_sha256']
    feature_path = LAB / 'rocbert-v9-inputs/features.pt'
    prepared = torch.load(feature_path, map_location='cpu', weights_only=True)
    policy, records = prepared['policy'], prepared['records']
    qualification_path = LAB / 'phonetic-views-v9-qualification-r3.json'
    qualification = json.loads(qualification_path.read_text(encoding='utf-8'))
    assert qualification['features_sha256'] == digest(feature_path)
    assert qualification['checked_training_views'] == 4745 and qualification['checked_evaluation_views'] == 3258
    assert qualification['source_lf_sha256'] == source_sha(LAB / 'rocbert_phonetic_view_v9.py')
    for name, value in policy['code_sha256'].items():
        assert source_sha(LAB / name) == value
    for name, value in policy['previous_3_10_policy']['evaluation_policy']['current_engine_sha256'].items():
        assert source_sha(ROOT / name) == value
    training = [(w, r['label'], r['id'], i) for r in records if r['cohort'] == 'training'
                for i, w in enumerate(r['windows'])]
    evaluation = [r for r in records if r['cohort'] != 'training']
    assert len(evaluation) == 3258 and len(training) == policy['training_windows']
    phonetics = PhoneticView(LAB / 'rocbert-base', LAB / 'rocbert-v8-inputs/mapping-qualification.json')
    by_id = {r['id']: r for r in records}
    checked_views = 0
    for row in records:
        if row.get('augmentation') == 'word_and_shape_unknown_keep_contextual_pronunciation':
            view, count = phonetics.apply(by_id[row['source_id']]['windows'])
            assert view == row['windows'] and count == row['masked_body_tokens']
            checked_views += 1
    assert checked_views == 4745
    model, base = create_model()
    state = load_file(str(parent_weights), device='cpu')
    missing, unexpected = model.load_state_dict(state, strict=True)
    assert not missing and not unexpected
    del state
    trainable = [p for p in model.parameters() if p.requires_grad]
    assert sum(p.numel() for p in trainable) == 117268994
    assert all(p.dtype == torch.float32 for p in trainable)
    optimizer = torch.optim.AdamW(trainable, lr=1e-5, weight_decay=.01, foreach=False)
    sources = [Path(__file__), LAB / 'train_rocbert_v8.py', LAB / 'rocbert_encoding_v9.py',
               LAB / 'rocbert_encoding_v8.py', LAB / 'phonetic_recheck_metrics.py',
               LAB / 'rocbert_phonetic_view_v9.py']
    code = {p.name: source_sha(p) for p in sources}
    metadata = dict(created_at_utc=datetime.now(UTC).isoformat(), dataset_policy=policy, base=base,
                    parent_weights_sha256=parent_sha, parent_epoch=3,
                    parent_choice='last fixed epoch, not selected on development scores',
                    features_sha256=digest(feature_path), index_sha256=digest(feature_path.with_name('index.json')),
                    code_sha256=code, trainable_parameters=117268994, epochs=1, seed=20261124,
                    optimizer='AdamW foreach=False', learning_rate=1e-5, weight_decay=.01,
                    gradient_clip=1., schedule='5% optimizer-step warmup then constant',
                    training_rows=policy['training_rows'], training_windows_per_epoch=len(training),
                    evaluation_rows=3258, excluded_evaluation_rows=0,
                    verified_training_phonetic_views=checked_views,
                    phonetic_view_qualification_sha256=digest(qualification_path),
                    evaluation_views='report both full input alone and fixed max(full,phonetic) for every row; all3258 retained',
                    training_time_budget_minutes=180, prediction_microbatch=16,
                    runtime=dict(torch=torch.__version__, cuda=torch.version.cuda,
                                 device=torch.cuda.get_device_name(0),
                                 packages={p: importlib.metadata.version(p) for p in
                                           ['transformers', 'tokenizers', 'safetensors', 'pypinyin', 'numpy']}),
                    production_acceptance=False)
    if probe:
        assert probe in {4, 8}
        worst = sorted(training, key=lambda x: len(x[0]['input_ids']), reverse=True)[:probe]
        model.train()
        torch.cuda.reset_peak_memory_stats()
        began = time.perf_counter()
        for stage in range(3):
            with torch.autocast('cuda', dtype=torch.bfloat16):
                logits = model(**inputs([w for w, _, _, _ in worst])).logits.float()
                loss = F.cross_entropy(logits, torch.tensor([y for _, y, _, _ in worst], device='cuda'))
            assert logits.shape == (probe, 2) and torch.isfinite(loss)
            loss.backward()
            if stage in {0, 2}:
                assert all(p.grad is not None and torch.isfinite(p.grad).all() for p in trainable)
                torch.nn.utils.clip_grad_norm_(trainable, 1.)
                optimizer.step()
                optimizer.zero_grad(set_to_none=True)
        metadata.update(micro_batch_size=probe, accumulate_steps=32 // probe, effective_batch=32,
                        probe_microbatches=3, probe_optimizer_updates=2,
                        probe_updates_reused=False, accuracy_evaluated=False,
                        peak_cuda_bytes=torch.cuda.max_memory_allocated(),
                        physical_total_bytes=torch.cuda.get_device_properties(0).total_memory,
                        elapsed_seconds=time.perf_counter() - began)
        metadata['physical_budget_pass'] = metadata['peak_cuda_bytes'] < .92 * metadata['physical_total_bytes']
        with (LAB / f'rocbert-v9-preflight-{probe}.json').open('x', encoding='utf-8') as handle:
            json.dump(metadata, handle, ensure_ascii=False, indent=2)
        print(json.dumps({k: metadata[k] for k in ['physical_budget_pass', 'peak_cuda_bytes',
                                                  'elapsed_seconds']}), flush=True)
        return 0 if metadata['physical_budget_pass'] else 2
    choices = []
    for size in [8, 4]:
        path = LAB / f'rocbert-v9-preflight-{size}.json'
        if path.exists():
            check = json.loads(path.read_text(encoding='utf-8'))
            assert check['features_sha256'] == metadata['features_sha256']
            assert check['parent_weights_sha256'] == parent_sha and check['code_sha256'] == code
            if check['physical_budget_pass']:
                choices.append(check)
    assert choices
    micro = choices[0]['micro_batch_size']
    accum = 32 // micro
    metadata.update(micro_batch_size=micro, accumulate_steps=accum, effective_batch=32,
                    preflight_sha256=digest(LAB / f'rocbert-v9-preflight-{micro}.json'),
                    probe_updates_reused=False)
    output = LAB / 'rocbert-v9'
    output.mkdir(exist_ok=False)
    rng = random.Random(20261125)
    order = list(range(len(training)))
    rng.shuffle(order)
    batches = []
    for start in range(0, len(order), 256):
        pool = sorted(order[start:start + 256], key=lambda i: len(training[i][0]['input_ids']))
        batches.extend(pool[i:i + micro] for i in range(0, len(pool), micro))
    rng.shuffle(batches)
    assert sorted(i for b in batches for i in b) == list(range(len(training)))
    with (output / 'training-window-order.json').open('x', encoding='utf-8') as handle:
        json.dump([batches], handle)
    metadata['training_order_sha256'] = digest(output / 'training-window-order.json')
    with (output / 'frozen.json').open('x', encoding='utf-8') as handle:
        json.dump(metadata, handle, ensure_ascii=False, indent=2)
    model.train()
    optimizer.zero_grad(set_to_none=True)
    began = time.perf_counter()
    deadline = time.monotonic() + 180 * 60
    warmup = max(1, math.ceil(math.ceil(len(batches) / accum) * .05))
    seen = 0
    total_loss = 0.
    update = 0
    for step, indices in enumerate(batches, 1):
        if time.monotonic() > deadline:
            raise TimeoutError('RoCBERT v9 exceeded180-minute training budget')
        batch = [training[i] for i in indices]
        group_start = ((step - 1) // accum) * accum
        group_count = sum(len(b) for b in batches[group_start:group_start + accum])
        with torch.autocast('cuda', dtype=torch.bfloat16):
            logits = model(**inputs([w for w, _, _, _ in batch])).logits.float()
            loss = F.cross_entropy(logits, torch.tensor([y for _, y, _, _ in batch], device='cuda'))
        if not torch.isfinite(loss):
            raise RuntimeError('Nonfinite phonetic modality loss')
        total_loss += float(loss.detach()) * len(batch)
        seen += len(batch)
        (loss * len(batch) / group_count).backward()
        if step % accum == 0 or step == len(batches):
            update += 1
            for group in optimizer.param_groups:
                group['lr'] = 1e-5 * min(1., update / warmup)
            torch.nn.utils.clip_grad_norm_(trainable, 1.)
            optimizer.step()
            optimizer.zero_grad(set_to_none=True)
            if update % 250 == 0:
                partial = output / f'partial-step-{step}'
                partial.mkdir(exist_ok=False)
                save(model, partial / 'model.safetensors')
                torch.save(dict(optimizer=optimizer.state_dict(), torch_rng=torch.get_rng_state(),
                                cuda_rng=torch.cuda.get_rng_state_all()), partial / 'optimizer-and-rng.pt')
                with (partial / 'progress.json').open('x', encoding='utf-8') as handle:
                    json.dump(dict(epoch=1, batches=step, total_batches=len(batches), windows_seen=seen,
                                   epoch_complete=False, accuracy_status='NOT_RUN', evaluation_run=False,
                                   features_sha256=metadata['features_sha256'], code_sha256=code), handle, indent=2)
        if step % 100 == 0:
            print(f'RoCBERT v9 batch={step}/{len(batches)} loss={total_loss/seen:.5f} '
                  f'elapsed={time.perf_counter()-began:.1f}s', flush=True)
    assert seen == len(training)
    training_seconds = time.perf_counter() - began
    directory = output / 'epoch-1'
    directory.mkdir(exist_ok=False)
    weights = directory / 'model.safetensors'
    save(model, weights)
    predictions, inference_seconds = predict(model, evaluation)
    with (directory / 'predictions.jsonl').open('x', encoding='utf-8') as handle:
        for row in predictions:
            handle.write(json.dumps(row, ensure_ascii=False) + '\n')
    result = dict(epoch=1, weights_sha256=digest(weights), training_seconds=training_seconds,
                  inference_seconds=inference_seconds, training_windows_seen=seen,
                  training_loss=total_loss/seen, evaluation_rows=3258, excluded_evaluation_rows=0,
                  peak_cuda_bytes=torch.cuda.max_memory_allocated(),
                  metrics=summarize(predictions, directory / 'thresholds.csv'), production_acceptance=False)
    with (directory / 'report.json').open('x', encoding='utf-8') as handle:
        json.dump(result, handle, ensure_ascii=False, indent=2)
    with (output / 'summary.json').open('x', encoding='utf-8') as handle:
        json.dump(dict(frozen=metadata, epochs=[result]), handle, ensure_ascii=False, indent=2)
    selected = result['metrics']['literal_or_model']['best_fpr_eligible']
    phonetic_records = []
    for row in evaluation:
        view, count = phonetics.apply(row['windows'])
        phonetic_records.append({**row, 'windows': view, 'phonetic_view_sha256': representation(view),
                                 'phonetic_masked_body_tokens': count})
    phonetic_predictions, phonetic_seconds = predict(model, phonetic_records)
    maximum_rows = []
    for raw, phone in zip(predictions, phonetic_predictions, strict=True):
        assert raw['id'] == phone['id'] and raw['label'] == phone['label']
        combined = {**raw, 'full_input_score': raw.get('score'), 'phonetic_input_score': phone.get('score'),
                    'phonetic_view_sha256': phone['phonetic_view_sha256'],
                    'phonetic_masked_body_tokens': phone['phonetic_masked_body_tokens']}
        if raw.get('error') or phone.get('error'):
            combined['error'] = 'FullOrPhoneticViewInferenceError'
        else:
            combined['score'] = max(raw['score'], phone['score'])
        maximum_rows.append(combined)
    alternate = output / 'epoch-1-phonetic-max'
    alternate.mkdir(exist_ok=False)
    with (alternate / 'predictions.jsonl').open('x', encoding='utf-8') as handle:
        for row in maximum_rows:
            handle.write(json.dumps(row, ensure_ascii=False) + '\n')
    alternate_result = dict(epoch=1, weights_sha256=result['weights_sha256'],
                            training_reused_from='epoch-1; no additional optimizer updates',
                            inference_seconds=inference_seconds + phonetic_seconds,
                            additional_phonetic_inference_seconds=phonetic_seconds,
                            evaluation_rows=3258, excluded_evaluation_rows=0,
                            view_aggregation='fixed max of both views for all3258 rows',
                            metrics=summarize(maximum_rows, alternate / 'thresholds.csv'),
                            production_acceptance=False)
    with (alternate / 'report.json').open('x', encoding='utf-8') as handle:
        json.dump(alternate_result, handle, ensure_ascii=False, indent=2)
    with (output / 'evaluation-complete.json').open('x', encoding='utf-8') as handle:
        json.dump(dict(frozen=metadata, full_input=result, phonetic_max=alternate_result),
                  handle, ensure_ascii=False, indent=2)
    maximum_selected = alternate_result['metrics']['literal_or_model']['best_fpr_eligible']
    print(json.dumps(dict(full_input_selected=selected, phonetic_max_selected=maximum_selected)), flush=True)
    return 0 if any(p and p['development_gate_met'] for p in [selected, maximum_selected]) else 1


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--probe-batch', type=int, default=0)
    raise SystemExit(main(parser.parse_args().probe_batch))
