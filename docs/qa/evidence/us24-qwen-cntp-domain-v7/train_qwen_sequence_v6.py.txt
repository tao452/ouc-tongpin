import argparse
import json
import math
import random
import time
from datetime import UTC, datetime
from pathlib import Path

import numpy as np
import torch
from safetensors.torch import save_file
from torch.nn import functional as F

from phonetic_recheck_metrics import summarize
from qwen_sequence_model_v5 import inputs
from qwen_sequence_model_v6 import create_model, digest, training_mode
from train_qwen_sequence_v5d import predict, source_sha

ROOT = Path.cwd()
LAB = ROOT / 'qa-private/us24-recall-20261124'


def optimizer_for(model):
    groups = []
    for head in [False, True]:
        parameters = [p for name, p in model.named_parameters() if p.requires_grad
                      and name.startswith('score.') == head]
        groups.append(dict(params=parameters, lr=1e-4 if head else 5e-6,
                           target_lr=1e-4 if head else 5e-6))
    return torch.optim.AdamW(groups, weight_decay=.01, foreach=False)


def save_weights(model, path):
    save_file({name: p.detach().cpu().contiguous() for name, p in model.named_parameters()
               if p.requires_grad}, str(path))


def main(probe_batch):
    assert torch.cuda.is_available() and torch.cuda.is_bf16_supported()
    torch.set_num_threads(4)
    random.seed(20261124)
    np.random.seed(20261124)
    torch.manual_seed(20261124)
    torch.cuda.manual_seed_all(20261124)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False
    torch.use_deterministic_algorithms(True, warn_only=True)
    features = LAB / 'qwen-homophone-v6-inputs/features.pt'
    prepared = torch.load(features, map_location='cpu', weights_only=True)
    policy, records = prepared['policy'], prepared['records']
    for name, value in policy['current_evaluation_policy']['current_engine_sha256'].items():
        assert source_sha(ROOT / name) == value
    training = [(window, row['label'], row['id'], n) for row in records if row['cohort']=='training'
                for n, window in enumerate(row['windows'])]
    evaluation = [r for r in records if r['cohort']!='training']
    assert len(evaluation)==3258 and sum(r['cohort']=='training' for r in records)==policy['training_rows']
    model, model_metadata = create_model()
    trainable = [p for p in model.parameters() if p.requires_grad]
    optimizer = optimizer_for(model)
    code = {p.name:source_sha(p) for p in [Path(__file__), LAB/'qwen_sequence_model_v6.py',
        LAB/'qwen_sequence_model_v5.py', LAB/'train_qwen_sequence_v5d.py', LAB/'phonetic_recheck_metrics.py']}
    metadata = dict(created_at_utc=datetime.now(UTC).isoformat(), dataset_policy=policy,
        features_sha256=digest(features), input_index_sha256=digest(features.with_name('index.json')),
        code_sha256=code, model=model_metadata,
        trainable_parameters={name:p.numel() for name,p in model.named_parameters() if p.requires_grad},
        total_trainable_parameters=sum(p.numel() for p in trainable),
        optimizer='AdamW foreach=False', learning_rates=dict(decoder_and_norm=5e-6,head=1e-4),
        schedule='5% optimizer-step linear warmup then constant', weight_decay=.01,
        gradient_clip=1., training_time_budget_minutes=180, training_rows=policy['training_rows'],
        training_windows=len(training), evaluation_rows=3258, excluded_evaluation_rows=0,
        seed=20261124, epochs=1, physical_device=torch.cuda.get_device_name(0),
        torch_version=torch.__version__, cuda_version=torch.version.cuda,
        production_acceptance=False)
    if probe_batch:
        assert probe_batch in {8,16}
        worst = sorted(training, key=lambda r:len(r[0]['input_ids']), reverse=True)[:probe_batch]
        training_mode(model)
        torch.cuda.reset_peak_memory_stats()
        began=time.perf_counter()
        with torch.autocast('cuda',dtype=torch.bfloat16):
            logits=model(**inputs([w for w,_,_,_ in worst])).logits.float()
            loss=F.cross_entropy(logits,torch.tensor([y for _,y,_,_ in worst],device='cuda'))
        assert logits.shape==(probe_batch,2) and torch.isfinite(loss)
        loss.backward()
        assert all(p.grad is not None and torch.isfinite(p.grad).all() for p in trainable)
        torch.nn.utils.clip_grad_norm_(trainable,1.)
        optimizer.step()  # Allocate actual Adam states; probe model is then discarded.
        optimizer.zero_grad(set_to_none=True)
        metadata.update(micro_batch_size=probe_batch, accumulate_steps=32//probe_batch,
            effective_batch=32, probe_optimizer_updates=1, probe_updates_reused=False,
            finite_loss=float(loss.detach()), elapsed_seconds=time.perf_counter()-began,
            peak_cuda_bytes=torch.cuda.max_memory_allocated(),
            physical_total_bytes=torch.cuda.get_device_properties(0).total_memory,
            accuracy_evaluated=False)
        metadata['physical_budget_pass']=metadata['peak_cuda_bytes']<.92*metadata['physical_total_bytes']
        with (LAB/f'qwen-sequence-v6-preflight-{probe_batch}.json').open('x',encoding='utf-8') as handle:
            json.dump(metadata,handle,ensure_ascii=False,indent=2)
        print(json.dumps({k:metadata[k] for k in ['micro_batch_size','peak_cuda_bytes','physical_budget_pass','elapsed_seconds','total_trainable_parameters']}),flush=True)
        return 0 if metadata['physical_budget_pass'] else 2
    probes=[]
    for size in [16,8]:
        path=LAB/f'qwen-sequence-v6-preflight-{size}.json'
        if path.exists():
            value=json.loads(path.read_text(encoding='utf-8'))
            assert value['features_sha256']==metadata['features_sha256'] and value['code_sha256']==code
            if value['physical_budget_pass']:probes.append(value)
    assert probes, 'A physical-memory qualified preflight is required'
    selected=probes[0]
    micro=selected['micro_batch_size'];accum=32//micro
    metadata.update(micro_batch_size=micro,accumulate_steps=accum,effective_batch=32,
        preflight_sha256=digest(LAB/f'qwen-sequence-v6-preflight-{micro}.json'),
        probe_updates_reused=False)
    output=LAB/'qwen-sequence-v6';output.mkdir(exist_ok=False)
    rng=random.Random(20261124);order=list(range(len(training)));rng.shuffle(order)
    batches=[]
    for start in range(0,len(order),256):
        pool=sorted(order[start:start+256],key=lambda i:len(training[i][0]['input_ids']))
        batches.extend(pool[i:i+micro] for i in range(0,len(pool),micro))
    rng.shuffle(batches)
    assert sorted(i for batch in batches for i in batch)==list(range(len(training)))
    with (output/'training-window-order.json').open('x',encoding='utf-8') as handle:json.dump(batches,handle)
    metadata['training_order_sha256']=digest(output/'training-window-order.json')
    with (output/'frozen.json').open('x',encoding='utf-8') as handle:json.dump(metadata,handle,ensure_ascii=False,indent=2)
    optimizer.zero_grad(set_to_none=True);training_mode(model)
    began=time.perf_counter();deadline=time.monotonic()+180*60
    warmup=max(1,math.ceil(math.ceil(len(batches)/accum)*.05));loss_sum=0.;seen=0
    torch.cuda.reset_peak_memory_stats()
    for step,indices in enumerate(batches,1):
        if time.monotonic()>deadline:raise TimeoutError('V6 exceeded180-minute training budget')
        batch=[training[i] for i in indices]
        group_start=((step-1)//accum)*accum
        group_count=sum(len(b) for b in batches[group_start:group_start+accum])
        with torch.autocast('cuda',dtype=torch.bfloat16):
            logits=model(**inputs([w for w,_,_,_ in batch])).logits.float()
            loss=F.cross_entropy(logits,torch.tensor([y for _,y,_,_ in batch],device='cuda'))
        if not torch.isfinite(loss):raise RuntimeError('Nonfinite V6 loss')
        loss_sum+=float(loss.detach())*len(batch);seen+=len(batch)
        (loss*len(batch)/group_count).backward()
        if step%accum==0 or step==len(batches):
            number=math.ceil(step/accum)
            for group in optimizer.param_groups:group['lr']=group['target_lr']*min(1.,number/warmup)
            torch.nn.utils.clip_grad_norm_(trainable,1.)
            optimizer.step();optimizer.zero_grad(set_to_none=True)
            if number%250==0:
                directory=output/f'partial-step-{step}';directory.mkdir(exist_ok=False)
                save_weights(model,directory/'parameters.safetensors')
                torch.save(dict(optimizer=optimizer.state_dict(),torch_rng=torch.get_rng_state(),
                    cuda_rng=torch.cuda.get_rng_state_all()),directory/'optimizer-and-rng.pt')
                with (directory/'progress.json').open('x',encoding='utf-8') as handle:
                    json.dump(dict(completed_batches=step,total_batches=len(batches),windows_seen=seen,
                        epoch_complete=False,accuracy_status='NOT_RUN',evaluation_run=False,
                        code_sha256=code,features_sha256=metadata['features_sha256']),handle,indent=2)
        if step%25==0:print(f'V6 batch={step}/{len(batches)} loss={loss_sum/seen:.5f} elapsed={time.perf_counter()-began:.1f}s',flush=True)
    assert seen==len(training)
    training_seconds=time.perf_counter()-began
    directory=output/'epoch-1';directory.mkdir(exist_ok=False)
    weights=directory/'parameters.safetensors';save_weights(model,weights)
    predictions,inference_seconds=predict(model,evaluation)
    with (directory/'predictions.jsonl').open('x',encoding='utf-8') as handle:
        for row in predictions:handle.write(json.dumps(row,ensure_ascii=False)+'\n')
    result=dict(epoch=1,weights_sha256=digest(weights),training_seconds=training_seconds,
        inference_seconds=inference_seconds,training_loss=loss_sum/seen,training_windows_seen=seen,
        peak_cuda_bytes=torch.cuda.max_memory_allocated(),evaluation_rows=len(predictions),excluded_evaluation_rows=0,
        metrics=summarize(predictions,directory/'thresholds.csv'),production_acceptance=False)
    with (directory/'report.json').open('x',encoding='utf-8') as handle:json.dump(result,handle,ensure_ascii=False,indent=2)
    with (output/'summary.json').open('x',encoding='utf-8') as handle:json.dump(dict(frozen=metadata,epochs=[result]),handle,ensure_ascii=False,indent=2)
    print(json.dumps(dict(best=result['metrics']['literal_or_model']['best_fpr_eligible'])),flush=True)
    return 0 if any(m['best_fpr_eligible'] and m['best_fpr_eligible']['development_gate_met'] for m in result['metrics'].values()) else 1


if __name__=='__main__':
    parser=argparse.ArgumentParser();parser.add_argument('--probe-batch',type=int,default=0)
    raise SystemExit(main(parser.parse_args().probe_batch))
