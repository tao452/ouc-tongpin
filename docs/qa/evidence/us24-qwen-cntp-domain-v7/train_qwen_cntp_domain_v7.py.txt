import argparse
import json
import math
import random
import time
from datetime import UTC,datetime
from pathlib import Path

import numpy as np
import torch
from safetensors.torch import load_file
from torch.nn import functional as F

from phonetic_recheck_metrics import summarize
from qwen_sequence_model_v5 import inputs
from qwen_sequence_model_v6 import create_model,digest,training_mode
from train_qwen_sequence_v5d import predict,source_sha
from train_qwen_sequence_v6 import optimizer_for,save_weights

ROOT=Path.cwd();LAB=ROOT/'qa-private/us24-recall-20261124'


def main(probe_batch):
    assert torch.cuda.is_available() and torch.cuda.is_bf16_supported()
    torch.set_num_threads(4);random.seed(20261124);np.random.seed(20261124)
    torch.manual_seed(20261124);torch.cuda.manual_seed_all(20261124)
    torch.backends.cudnn.deterministic=True;torch.backends.cudnn.benchmark=False
    torch.use_deterministic_algorithms(True,warn_only=True)
    features=LAB/'qwen-cntp-domain-v7-inputs/features.pt'
    prepared=torch.load(features,map_location='cpu',weights_only=True)
    policy,records=prepared['policy'],prepared['records']
    for name,value in policy['current_evaluation_policy']['current_engine_sha256'].items():assert source_sha(ROOT/name)==value
    training=[(w,r['label'],r['id'],i) for r in records if r['cohort']=='training' for i,w in enumerate(r['windows'])]
    evaluation=[r for r in records if r['cohort']!='training'];assert len(evaluation)==3258
    model,model_metadata=create_model()
    warm=LAB/'qwen-sequence-v6/epoch-1/parameters.safetensors'
    previous=json.loads((warm.parent/'report.json').read_text(encoding='utf-8'))
    assert digest(warm)==previous['weights_sha256']
    state=load_file(str(warm),device='cpu');parameters=dict(model.named_parameters())
    assert set(state)=={name for name,p in parameters.items() if p.requires_grad}
    with torch.no_grad():
        for name,value in state.items():parameters[name].copy_(value)
    del state
    trainable=[p for p in model.parameters() if p.requires_grad]
    assert sum(p.numel() for p in trainable)==100792832
    optimizer=optimizer_for(model)
    code={p.name:source_sha(p) for p in [Path(__file__),LAB/'qwen_sequence_model_v6.py',
        LAB/'qwen_sequence_model_v5.py',LAB/'train_qwen_sequence_v5d.py',LAB/'train_qwen_sequence_v6.py',LAB/'phonetic_recheck_metrics.py']}
    metadata=dict(created_at_utc=datetime.now(UTC).isoformat(),dataset_policy=policy,
        features_sha256=digest(features),index_sha256=digest(features.with_name('index.json')),
        code_sha256=code,model=model_metadata,warm_start_v6_weights_sha256=digest(warm),
        total_trainable_parameters=100792832,epochs=2,seed=20261124,
        optimizer='AdamW foreach=False',learning_rates=dict(decoder_and_norm=5e-6,head=1e-4),
        schedule='5% total2epoch optimizer-step warmup then constant',weight_decay=.01,gradient_clip=1.,
        training_rows=policy['training_rows'],training_windows_per_epoch=len(training),
        evaluation_rows=3258,excluded_evaluation_rows=0,training_time_budget_minutes=180,
        torch_version=torch.__version__,cuda_version=torch.version.cuda,production_acceptance=False)
    if probe_batch:
        assert probe_batch in {8,16}
        worst=sorted(training,key=lambda r:len(r[0]['input_ids']),reverse=True)[:probe_batch]
        torch.cuda.reset_peak_memory_stats();training_mode(model);began=time.perf_counter()
        with torch.autocast('cuda',dtype=torch.bfloat16):
            logits=model(**inputs([w for w,_,_,_ in worst])).logits.float()
            loss=F.cross_entropy(logits,torch.tensor([y for _,y,_,_ in worst],device='cuda'))
        assert logits.shape==(probe_batch,2) and torch.isfinite(loss)
        loss.backward();assert all(p.grad is not None and torch.isfinite(p.grad).all() for p in trainable)
        torch.nn.utils.clip_grad_norm_(trainable,1.);optimizer.step();optimizer.zero_grad(set_to_none=True)
        metadata.update(micro_batch_size=probe_batch,accumulate_steps=32//probe_batch,effective_batch=32,
            probe_optimizer_updates=1,probe_updates_reused=False,accuracy_evaluated=False,
            peak_cuda_bytes=torch.cuda.max_memory_allocated(),physical_total_bytes=torch.cuda.get_device_properties(0).total_memory,
            elapsed_seconds=time.perf_counter()-began,finite_loss=float(loss.detach()))
        metadata['physical_budget_pass']=metadata['peak_cuda_bytes']<.92*metadata['physical_total_bytes']
        with (LAB/f'qwen-cntp-domain-v7-preflight-{probe_batch}.json').open('x',encoding='utf-8') as f:json.dump(metadata,f,ensure_ascii=False,indent=2)
        print(json.dumps({k:metadata[k] for k in ['physical_budget_pass','peak_cuda_bytes','elapsed_seconds']}),flush=True)
        return 0 if metadata['physical_budget_pass'] else 2
    probes=[]
    for size in [16,8]:
        p=LAB/f'qwen-cntp-domain-v7-preflight-{size}.json'
        if p.exists():
            value=json.loads(p.read_text(encoding='utf-8'))
            assert value['features_sha256']==metadata['features_sha256'] and value['code_sha256']==code
            if value['physical_budget_pass']:probes.append(value)
    assert probes
    selected=probes[0];micro=selected['micro_batch_size'];accum=32//micro
    metadata.update(micro_batch_size=micro,accumulate_steps=accum,effective_batch=32,probe_updates_reused=False,
        preflight_sha256=digest(LAB/f'qwen-cntp-domain-v7-preflight-{micro}.json'))
    output=LAB/'qwen-cntp-domain-v7';output.mkdir(exist_ok=False)
    orders=[]
    for epoch in [1,2]:
        rng=random.Random(20261124+epoch);order=list(range(len(training)));rng.shuffle(order);batches=[]
        for start in range(0,len(order),256):
            pool=sorted(order[start:start+256],key=lambda i:len(training[i][0]['input_ids']))
            batches.extend(pool[i:i+micro] for i in range(0,len(pool),micro))
        rng.shuffle(batches);assert sorted(i for b in batches for i in b)==list(range(len(training)))
        orders.append(batches)
    with (output/'training-window-order.json').open('x',encoding='utf-8') as f:json.dump(orders,f)
    metadata['training_order_sha256']=digest(output/'training-window-order.json')
    with (output/'frozen.json').open('x',encoding='utf-8') as f:json.dump(metadata,f,ensure_ascii=False,indent=2)
    deadline=time.monotonic()+180*60;warmup=max(1,math.ceil(sum(math.ceil(len(b)/accum) for b in orders)*.05))
    update=0;results=[]
    for epoch,batches in enumerate(orders,1):
        training_mode(model);optimizer.zero_grad(set_to_none=True);began=time.perf_counter();loss_sum=0.;seen=0
        for step,indices in enumerate(batches,1):
            if time.monotonic()>deadline:raise TimeoutError('V7 exceeded180-minute combined training budget')
            batch=[training[i] for i in indices];group_start=((step-1)//accum)*accum
            group_count=sum(len(b) for b in batches[group_start:group_start+accum])
            with torch.autocast('cuda',dtype=torch.bfloat16):
                logits=model(**inputs([w for w,_,_,_ in batch])).logits.float()
                loss=F.cross_entropy(logits,torch.tensor([y for _,y,_,_ in batch],device='cuda'))
            if not torch.isfinite(loss):raise RuntimeError('NonfiniteV7 training loss')
            loss_sum+=float(loss.detach())*len(batch);seen+=len(batch)
            (loss*len(batch)/group_count).backward()
            if step%accum==0 or step==len(batches):
                update+=1
                for group in optimizer.param_groups:group['lr']=group['target_lr']*min(1.,update/warmup)
                torch.nn.utils.clip_grad_norm_(trainable,1.);optimizer.step();optimizer.zero_grad(set_to_none=True)
                if update%250==0:
                    partial=output/f'partial-epoch-{epoch}-step-{step}';partial.mkdir(exist_ok=False)
                    save_weights(model,partial/'parameters.safetensors')
                    torch.save(dict(optimizer=optimizer.state_dict(),torch_rng=torch.get_rng_state(),cuda_rng=torch.cuda.get_rng_state_all()),partial/'optimizer-and-rng.pt')
                    with (partial/'progress.json').open('x',encoding='utf-8') as f:json.dump(dict(epoch=epoch,completed_batches=step,total_batches=len(batches),windows_seen=seen,epoch_complete=False,accuracy_status='NOT_RUN',evaluation_run=False,code_sha256=code),f,indent=2)
            if step%25==0:print(f'V7 epoch={epoch} batch={step}/{len(batches)} loss={loss_sum/seen:.5f} elapsed={time.perf_counter()-began:.1f}s',flush=True)
        assert seen==len(training);training_seconds=time.perf_counter()-began
        directory=output/f'epoch-{epoch}';directory.mkdir(exist_ok=False)
        weights=directory/'parameters.safetensors';save_weights(model,weights)
        predictions,inference_seconds=predict(model,evaluation)
        with (directory/'predictions.jsonl').open('x',encoding='utf-8') as f:
            for row in predictions:f.write(json.dumps(row,ensure_ascii=False)+'\n')
        result=dict(epoch=epoch,weights_sha256=digest(weights),training_seconds=training_seconds,inference_seconds=inference_seconds,
            training_loss=loss_sum/seen,training_windows_seen=seen,evaluation_rows=len(predictions),excluded_evaluation_rows=0,
            peak_cuda_bytes=torch.cuda.max_memory_allocated(),metrics=summarize(predictions,directory/'thresholds.csv'),production_acceptance=False)
        with (directory/'report.json').open('x',encoding='utf-8') as f:json.dump(result,f,ensure_ascii=False,indent=2)
        results.append(result)
        with (output/f'summary-through-epoch-{epoch}.json').open('x',encoding='utf-8') as f:json.dump(dict(frozen=metadata,epochs=results),f,ensure_ascii=False,indent=2)
        print(json.dumps(dict(epoch=epoch,best=result['metrics']['literal_or_model']['best_fpr_eligible'])),flush=True)
    with (output/'summary.json').open('x',encoding='utf-8') as f:json.dump(dict(frozen=metadata,epochs=results),f,ensure_ascii=False,indent=2)
    return 0 if any(m['best_fpr_eligible'] and m['best_fpr_eligible']['development_gate_met'] for r in results for m in r['metrics'].values()) else 1


if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--probe-batch',type=int,default=0)
    raise SystemExit(main(p.parse_args().probe_batch))
