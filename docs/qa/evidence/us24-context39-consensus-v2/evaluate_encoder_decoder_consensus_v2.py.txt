import hashlib
import json
import math
from datetime import UTC,datetime
from pathlib import Path

from phonetic_recheck_metrics import summarize

ROOT=Path.cwd();LAB=ROOT/'qa-private/us24-recall-20261124'
paths=[LAB/'phonetic-encoder-context39-r2/predictions.jsonl',LAB/'qwen-sequence-v5d/epoch-1/predictions.jsonl']
sources=[{r['id']:r for r in map(json.loads,p.read_text(encoding='utf-8').splitlines())} for p in paths]
assert len(sources[0])==len(sources[1])==3258 and sources[0].keys()==sources[1].keys()
for identifier in sources[0]:
    for key in ['label','cohort','category','source_text_sha256','normalized_sha256','transformed_sha256','literal_any']:
        assert sources[0][identifier][key]==sources[1][identifier][key],(identifier,key)
output=LAB/'encoder-decoder-consensus-v2';output.mkdir(exist_ok=False)
frozen=dict(created_at_utc=datetime.now(UTC).isoformat(),scope='fixed two-model development combination; no per-row/category switching',
    methods=['equal_probability_mean','equal_logit_mean','maximum_score_diagnostic'],
    sources=[dict(path=p.relative_to(LAB).as_posix(),sha256=hashlib.sha256(p.read_bytes()).hexdigest()) for p in paths],
    code_sha256=hashlib.sha256(Path(__file__).read_bytes().replace(b'\r\n',b'\n')).hexdigest(),
    evaluation_rows=3258,excluded_evaluation_rows=0,COLD_dev_fetched=False,production_acceptance=False)
with (output/'frozen.json').open('x',encoding='utf-8') as handle:json.dump(frozen,handle,ensure_ascii=False,indent=2)
results={}
for method in frozen['methods']:
    directory=output/method;directory.mkdir(exist_ok=False);predictions=[]
    for identifier,first in sources[0].items():
        rows=[source[identifier] for source in sources]
        item={k:v for k,v in first.items() if k not in {'score','error'}}
        if any(r.get('error') or not math.isfinite(r.get('score',float('nan'))) for r in rows):item['error']='AnySourceInferenceError'
        else:
            scores=[r['score'] for r in rows]
            if method=='equal_probability_mean':value=sum(scores)/2
            elif method=='maximum_score_diagnostic':value=max(scores)
            else:
                scores=[min(1-1e-6,max(1e-6,p)) for p in scores]
                value=1/(1+math.exp(-sum(math.log(p/(1-p)) for p in scores)/2))
            item['score']=value
        predictions.append(item)
    with (directory/'predictions.jsonl').open('x',encoding='utf-8') as handle:
        for row in predictions:handle.write(json.dumps(row,ensure_ascii=False)+'\n')
    report=dict(method=method,evaluation_rows=3258,excluded_evaluation_rows=0,
        metrics=summarize(predictions,directory/'thresholds.csv'),production_acceptance=False)
    with (directory/'report.json').open('x',encoding='utf-8') as handle:json.dump(report,handle,ensure_ascii=False,indent=2)
    results[method]=report
    print(json.dumps(dict(method=method,best=report['metrics']['literal_or_model']['best_fpr_eligible'])),flush=True)
with (output/'summary.json').open('x',encoding='utf-8') as handle:json.dump(dict(frozen=frozen,results=results),handle,ensure_ascii=False,indent=2)
raise SystemExit(0 if any(m['best_fpr_eligible'] and m['best_fpr_eligible']['development_gate_met'] for r in results.values() for m in r['metrics'].values()) else 1)
